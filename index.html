-- ========================================================= -- DATABASE ONLINE TOOL KONTROL PENJUALAN -- VERSI PERBAIKAN (policy admin_users & casting auth.uid()::uuid) -- ========================================================= create extension if not exists pgcrypto; -- ========================================================= -- 1. TABEL PRODUCTS -- ========================================================= create table if not exists public.products ( id uuid primary key default gen_random_uuid(), name text not null, price numeric(14,2) not null default 0 check (price >= 0), active boolean not null default true, created_at timestamptz not null default now() ); -- Index nama produk (menggunakan lower(name) agar case-insensitive) create unique index if not exists products_name_unique on public.products (lower(name)); -- ========================================================= -- 2. TABEL SETTINGS -- ========================================================= create table if not exists public.settings ( id integer primary key default 1, target_pcs integer not null default 0 check (target_pcs >= 0), target_omzet numeric(14,2) not null default 0 check (target_omzet >= 0), updated_at timestamptz not null default now() ); -- Pastikan baris settings tersedia insert into public.settings (id, target_pcs, target_omzet) select 1, 0, 0 where not exists ( select 1 from public.settings where id = 1 ); -- ========================================================= -- 3. TABEL ADMIN USERS -- ========================================================= create table if not exists public.admin_users ( user_id uuid primary key references auth.users(id) on delete cascade, created_at timestamptz not null default now() ); -- ========================================================= -- 4. TABEL SALES -- ========================================================= create table if not exists public.sales ( id uuid primary key default gen_random_uuid(), cashier_user_id uuid references auth.users(id) on delete set null, cashier_name text, shift text, product_id uuid references public.products(id) on delete set null, product_name text not null, unit_price numeric(14,2) not null default 0 check (unit_price >= 0), qty integer not null default 1 check (qty > 0), total numeric(14,2) not null default 0 check (total >= 0), sold_at timestamptz not null default now() ); -- ========================================================= -- 5. INDEX PENJUALAN -- ========================================================= create index if not exists sales_sold_at_idx on public.sales (sold_at); create index if not exists sales_cashier_name_idx on public.sales (cashier_name); create index if not exists sales_product_id_idx on public.sales (product_id); -- ========================================================= -- 6. FUNCTION OTOMATIS HARGA PENJUALAN -- ========================================================= create or replace function public.validate_sale() returns trigger language plpgsql security definer set search_path = public as $$ declare p_name text; p_price numeric(14,2); begin -- Jika product_id digunakan if new.product_id is not null then select name, price into p_name, p_price from public.products where id = new.product_id and active = true; -- Produk harus aktif if p_name is null then raise exception 'Produk tidak ditemukan atau sudah tidak aktif'; end if; -- Ambil harga terbaru dari database new.product_name := p_name; new.unit_price := p_price; end if; -- Hitung total otomatis new.total := round(new.unit_price * new.qty, 2); return new; end; $$; -- NOTE: Fungsi ini dibuat SECURITY DEFINER. Pastikan owner fungsi adalah role yang benar -- (service account / db owner) dan bukan role dengan privilege terbatas. Sesuaikan owner -- jika perlu, mis.: -- alter function public.validate_sale() owner to postgres; -- sesuaikan role target -- ========================================================= -- 7. TRIGGER PENJUALAN -- ========================================================= drop trigger if exists sales_validate_before_insert on public.sales; create trigger sales_validate_before_insert before insert on public.sales for each row execute function public.validate_sale(); -- ========================================================= -- 8. AKTIFKAN RLS -- ========================================================= alter table public.products enable row level security; alter table public.settings enable row level security; alter table public.admin_users enable row level security; alter table public.sales enable row level security; -- ========================================================= -- 9. HAPUS POLICY LAMA -- ========================================================= drop policy if exists "Public can view active products" on public.products; drop policy if exists "Admins can manage products" on public.products; drop policy if exists "Admins can manage settings" on public.settings; drop policy if exists "Admins can manage admin users" on public.admin_users; drop policy if exists "Anyone can insert sales" on public.sales; drop policy if exists "Admins can read sales" on public.sales; drop policy if exists "Admins can delete sales" on public.sales; -- ========================================================= -- 10. POLICY PRODUCTS -- ========================================================= create policy "Public can view active products" on public.products for select to anon, authenticated using (active = true); create policy "Admins can manage products" on public.products for all to authenticated using ( exists ( select 1 from public.admin_users where user_id = auth.uid()::uuid ) ) with check ( exists ( select 1 from public.admin_users where user_id = auth.uid()::uuid ) ); -- ========================================================= -- 11. POLICY SETTINGS -- ========================================================= create policy "Admins can manage settings" on public.settings for all to authenticated using ( exists ( select 1 from public.admin_users where user_id = auth.uid()::uuid ) ) with check ( exists ( select 1 from public.admin_users where user_id = auth.uid()::uuid ) ); -- ========================================================= -- 12. POLICY ADMIN USERS -- ========================================================= -- PENTING: kebijakan ini TIDAK mengizinkan self-insert. Hanya user yang sudah menjadi admin -- (ada di public.admin_users) yang dapat mengelola admin_users. create policy "Admins can manage admin users" on public.admin_users for all to authenticated using ( exists ( select 1 from public.admin_users where user_id = auth.uid()::uuid ) ) with check ( exists ( select 1 from public.admin_users where user_id = auth.uid()::uuid ) ); -- ========================================================= -- 13. POLICY SALES -- ========================================================= create policy "Anyone can insert sales" on public.sales for insert to anon, authenticated with check (true); create policy "Admins can read sales" on public.sales for select to authenticated using ( exists ( select 1 from public.admin_users where user_id = auth.uid()::uuid ) ); create policy "Admins can delete sales" on public.sales for delete to authenticated using ( exists ( select 1 from public.admin_users where user_id = auth.uid()::uuid ) ); -- ========================================================= -- 14. IZIN AKSES -- ========================================================= grant usage on schema public to anon, authenticated; grant select on public.products to anon, authenticated; grant insert on public.sales to anon, authenticated; grant select, insert, update, delete on public.sales to authenticated; grant select, insert, update on public.settings to authenticated; grant select, insert, update, delete on public.products to authenticated; grant select, insert, update, delete on public.admin_users to authenticated; -- ========================================================= -- 15.
